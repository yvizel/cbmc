27/*
28 * A horn clause.
29 * This class is simply a wrapper around a forall_exprt with a few utilities:
30 * 1. Getting the body of a clause
31 * 2. Getting the head of a clause
32 * 3. Checking if a clause is a fact or a query
33 * 4. Getting used relations (the predicates) or function applications
34 * (their instantiations) in a clause.
35 */
36class horn_clauset
37{
38 forall_exprt m_chc;
39
40public:
41 explicit horn_clauset(forall_exprt f) : m_chc(f)
42 {
43 }
44
45 horn_clauset(std::vector<symbol_exprt> &vars, exprt clause)
46 : m_chc(vars, clause)
47 {
48 }
49
50 const forall_exprt &get_chc() const
51 {
52 return m_chc;
53 }
54
55 const exprt *body() const
56 {
57 if(can_cast_expr<implies_exprt>(m_chc.where()))
58 {
59 return &to_implies_expr(m_chc.where()).op0();
60 }
61 return &m_chc.where();
62 }
63
64 const exprt *head() const
65 {
66 if(can_cast_expr<implies_exprt>(m_chc.where()))
67 {
68 return &to_implies_expr(m_chc.where()).op1();
69 }
70 return nullptr;
71 }
72
73 bool is_fact() const
74 {
75 auto b = body();
76 const std::function<bool(const exprt &)> pred = [](const exprt &subexpr)
77 { return can_cast_expr<function_application_exprt>(subexpr); };
78 auto it = std::find_if(b->depth_begin(), b->depth_end(), pred);
79
80 return (it == b->depth_end());
81 }
82
83 bool is_query() const
84 {
85 if(can_cast_expr<implies_exprt>(m_chc.where()))
86 {
87 return !(can_cast_expr<function_application_exprt>(*head()));
88 }
89 return false;
90 }
91
92 bool operator==(const horn_clauset &other) const
93 {
94 return m_chc == other.m_chc;
95 }
96
97 bool operator!=(const horn_clauset &other) const
98 {
99 return !(*this == other);
100 }
101
102 bool operator<(const horn_clauset &other) const
103 {
104 return m_chc < other.m_chc;
105 }
106
107 std::vector<symbol_exprt> used_relations(chc_dbt &db) const;
108 void used_func_app(chc_dbt &db, std::vector<function_application_exprt> &out)
109 const;
110};

172 auto it = m_body_idx.find(state);

174 return m_empty_set;

176 }
177
178 const chc_sett &def(const exprt &state) const
179 {
180 auto it = m_head_idx.find(state);
181 if(it == m_head_idx.end())
182 return m_empty_set;
183 return it->second;
184 }
185
186 void add_clause(const forall_exprt &f)
187 {
188 if(f.is_true())
189 return;
190 auto new_cls = horn_clauset(f);
191 // Equivalent (semantic) queries may represent
192 // different properties
193 if(!new_cls.is_query())
194 {
195 for(auto &c : m_clauses)
196 {
197 if(c.get_chc() == f)
198 return;
199 }
200 }
201 m_clauses.push_back(new_cls);
202 reset_indices();
203 }
204
205 [[nodiscard]] const horn_clauset &get_clause(std::size_t idx) const
206 {
207 INVARIANT(idx < m_clauses.size(), "Index in range");
208 return m_clauses[idx];
209 }
210
211 chcst::iterator begin()
212 {

214 }
215 chcst::iterator end()
216 {
217 return m_clauses.end();
218 }
219 chcst::const_iterator begin() const
220 {
221 return m_clauses.begin();
222 }
223 chcst::const_iterator end() const
224 {
225 return m_clauses.end();
226 }
