66 // Create the new CHC db after eliminating uninterpreted predicates.
67 void populate_new_db()
68 {
69 std::vector<symbol_exprt> rels;
70 for(auto &clause : m_db)
71 {
72 if(clause.is_query())
73 {
74 std::vector<symbol_exprt> r = clause.used_relations(m_db);
75 rels.insert(rels.end(), r.begin(), r.end());

80 for(auto p : rels)

83 }

86 {
87 auto c = m_def.find(it);
88 INVARIANT(c != m_def.end(), "No clauses");
89 for(auto clause : c->second)
90 m_new_db.add_clause(clause.get_chc());
91 }
92
93 for(auto &clause : m_db)
94 {
95 if(clause.is_query())
96 {
97 m_new_db.add_clause(clause.get_chc());
98 }
99 }

103 void eliminate(const symbol_exprt *symb)
104 {
105 for(auto idx : m_db.def(*symb))
106 {
107 auto &clause = m_db.get_clause(idx);
108 std::vector<symbol_exprt> use = clause.used_relations(m_db);
109 if(use.size() > 1)
110 {
111 throw incorrect_goto_program_exceptiont(
112 "Non-linear CHCs not supported yet");
113 }
114
115 if(clause.is_fact())
116 {
117 m_heads.insert(symb->hash());
118 std::vector<horn_clauset> def_chcs;
119 def_chcs.push_back(clause);
120 m_def.insert(std::make_pair(symb->hash(), def_chcs));
121 }
122
123 for(auto &p : use)
124 {
125 auto it = m_def.find(p.hash());
126 std::vector<horn_clauset> def_chcs;
127 if(it == m_def.end() || m_heads.find(p.hash()) != m_heads.end())
128 {
129 def_chcs.push_back(clause);
130 }
131 else
132 {
133 for(auto cls_it = it->second.begin(); cls_it != it->second.end();
134 cls_it++)
135 {
136 forall_exprt resolvent = resolve_clauses((*cls_it), clause);
137 if(m_verbose)
138 std::cout << "Result:\n" << format(resolvent) << "\n";
139 def_chcs.push_back(horn_clauset(resolvent));
140 }
141 }
142 auto def_it = m_def.find(symb->hash());
143 if(def_it == m_def.end())
144 m_def.insert(std::make_pair(symb->hash(), def_chcs));
145 else
146 {
147 auto &vec = def_it->second;
148 vec.insert(vec.end(), def_chcs.begin(), def_chcs.end());
149 }
150 }
151 }

161 forall_exprt resolve_clauses(const horn_clauset &c1, const horn_clauset &c2)
162 {
163 const exprt &body1 = *c1.body();
164 const exprt &head1 = *c1.head();
165 const exprt &body2 = *c2.body();
166 const exprt &head2 = *c2.head();
167
168 std::vector<function_application_exprt> use2;
169 c2.used_func_app(m_db, use2);
170
171 INVARIANT(use2.size() == 1, "Only handling linear case");
172 if(use2.size() != 1)
173 throw analysis_exceptiont("Resolution not possible");
174
175 function_application_exprt &body2_pred = use2[0];
176
177 const function_application_exprt *head1_pred = nullptr;
178 if(can_cast_expr<function_application_exprt>(*c1.head()))
179 {
180 head1_pred = &to_function_application_expr(head1);
181 }
182 if(head1_pred == nullptr)
183 throw analysis_exceptiont("Resolution not possible");
184
185 if(m_verbose)
186 std::cout << "Resolving: \n"
187 << format(c1.get_chc()) << "\nAnd: \n"
188 << format(c2.get_chc()) << "\n";
189
190 std::set<symbol_exprt> all_vars(
191 c1.get_chc().variables().begin(), c1.get_chc().variables().end());
192 all_vars.insert(
193 c2.get_chc().variables().begin(), c2.get_chc().variables().end());
194
195 const function_application_exprt *body1_pred = nullptr;
196 exprt::operandst body_ops;
197 if(body1.id() == ID_and)
198 {
199 body1_pred = &to_function_application_expr(to_and_expr(body1).op0());
200 body_ops.push_back(to_and_expr(body1).op1());
201 }
202 else
203 {
204 body1_pred = &to_function_application_expr(body1);
205 }
206 exprt transformed_body = (can_cast_expr<and_exprt>(body2))
207 ? to_and_expr(body2).op1()
208 : true_exprt();
209 exprt transformed_head = head2;
210
211 INVARIANT(
212 head1_pred->arguments().size() == 1,
213 "Only one argument to predicate is assumes");
214 const exprt &head_arg = head1_pred->arguments().at(0);
215 const symbol_exprt &body_arg = to_symbol_expr(body2_pred.arguments().at(0));
216
217 if(
218 (head_arg.id() != ID_symbol) ||
219 (to_symbol_expr(head_arg).get_identifier() != body_arg.get_identifier()))
220 {
221 std::map<irep_idt, exprt> subs;
222 subs.insert(std::make_pair(body_arg.get_identifier(), head_arg));
223
224 if(!transformed_body.is_true())
225 {
226 std::optional<exprt> s = substitute_symbols(subs, transformed_body);
227 if(s.has_value())
228 transformed_body = std::move(s.value());
229 }
230
231 std::optional<exprt> s = substitute_symbols(subs, transformed_head);
232 if(s.has_value())
233 {
234 transformed_head = std::move(s.value());
235 }
236 }
237
238 body_ops.push_back(transformed_body);
239 transformed_body = and_exprt(*body1_pred, std::move(and_exprt(body_ops)));
240
241 forall_exprt f(
242 std::vector<symbol_exprt>(all_vars.begin(), all_vars.end()),
243 implies_exprt(std::move(transformed_body), std::move(transformed_head)));
244 return f;
245 }
