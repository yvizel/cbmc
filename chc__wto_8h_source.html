200 typedef std::shared_ptr<wto_elementt> wto_element_ptrt;
201 typedef std::deque<wto_element_ptrt> partitiont;
202 typedef std::unordered_map<std::size_t, std::vector<wto_componentt>>
203 nested_componentst;

205 chc_grapht &m_g;
206
207 std::unordered_map<std::size_t, inf_numt> m_dfn;
208 std::vector<const symbol_exprt *> m_stack;
209 unsigned m_cur_dfn_num;
210
211 std::deque<wto_element_ptrt> m_partition;
212 nested_componentst m_nested_comp;
213
214private:
215 // helper to avoid initializing all vertices to zero
216 inf_numt get_dfn(const symbol_exprt *v) const
217 {
218 auto it = m_dfn.find(v->hash());
219 if(it != m_dfn.end())
220 return it->second;
221 return inf_numt(0);
222 }
223
224 std::deque<wto_element_ptrt> component(const symbol_exprt *v)
225 {
226 std::deque<wto_element_ptrt> partition;
227 for(auto &target : m_g.outgoing(*v))
228 {
229 const symbol_exprt *t = (&to_symbol_expr(target));
230 if(get_dfn(t) == inf_numt(0))
231 visit(t, partition);
232 }
233 return partition;
234 }
235
236 inf_numt visit(const symbol_exprt *v, std::deque<wto_element_ptrt> &partition)
237 {
238 m_stack.push_back(v);
239 m_dfn[v->hash()] = inf_numt(m_cur_dfn_num++);
240 auto head = get_dfn(v);
241 bool loop = false;
242 for(auto &target : m_g.outgoing(*v))
243 {
244 const symbol_exprt *t = (&to_symbol_expr(target));
245 auto min = get_dfn(t);
246 if(min == inf_numt(0))
247 min = visit(t, partition);
248 if(min <= head)
249 {
250 head = min;
251 loop = true;
252 }
253 }
254
255 if(head == get_dfn(v))
256 { // v is the head of a component
257 m_dfn[v->hash()] = inf_numt();
258 auto element = m_stack.back();
259 m_stack.pop_back();
260 if(loop)
261 {
262 while(!(element == v))
263 {
264 m_dfn[element->hash()] = inf_numt(0); // reset
265 element = m_stack.back();
266 m_stack.pop_back();
267 }
268 partition.push_front(std::static_pointer_cast<wto_elementt>(
269 std::make_shared<wto_componentt>(v, component(v))));
270 }
271 else
272 {
273 partition.push_front(std::static_pointer_cast<wto_elementt>(
274 std::make_shared<wto_singletont>(v)));
275 }
276 }
277 return head;
278 }
