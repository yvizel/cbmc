99void cutpoint_grapht::compute_edges(const goto_functiont &goto_function)
100{
101 forall_goto_program_instructions(it, goto_function.body)
102 {
103 if(is_cutpoint(*it))
104 {
105 std::vector<bool> &reach = m_fwd[&(*it)];
106 cutpointt &cp = get_cutpoint(*it);
107
108 for(std::size_t r = 0; r < reach.size(); r++)
109 {
110 if(reach[r] == true)
111 {
112 cutpoint_edge_ptrt edg = create_edge(cp, *m_cps[r]);
113 edg->push_back(&(*it));
114 }
115 }
116 }
117 else
118 {
119 std::vector<bool> &breach = m_bwd[&(*it)];
120 std::vector<bool> &freach = m_fwd[&(*it)];
121
122 for(std::size_t i = 0; i < breach.size(); i++)
123 {
124 if(breach[i] == false)
125 continue;
126 for(std::size_t j = 0; j < freach.size(); j++)
127 {
128 if(freach[j] == false)
129 continue;
130 cutpoint_edge_ptrt edge = get_edge(*m_cps[i], *m_cps[j]);
131 edge->push_back(&(*it));
132 }
133 }
134 }
135 }

138void cutpoint_grapht::compute_cutpoints(const goto_functiont &goto_function)
139{
140 m_cps.clear();
141
142 std::map<const goto_programt::instructiont *, std::size_t> cp_map;
143
144 forall_goto_program_instructions(it, goto_function.body)
145 {
146 if(cp_map.find(&*it) != cp_map.end())
147 continue;
148
149 // entry
150 if(it->incoming_edges.empty())
151 {
152 auto i = cp_map.find(&*it);
153 if(i == cp_map.end())
154 cp_map.insert(std::make_pair(&*it, cp_map.size()));
155 }
156
157 // exit
158 if(it->is_end_function())
159 {
160 auto i = cp_map.find(&*it);
161 if(i == cp_map.end())
162 cp_map.insert(std::make_pair(&*it, cp_map.size()));
163 }
164
165 for(auto in : it->incoming_edges)
166 {
167 auto &inst = *in;
168 if(inst.is_backwards_goto())
169 {
170 auto i = cp_map.find(&*it);
171 if(i == cp_map.end())
172 cp_map.insert(std::make_pair(&*it, cp_map.size()));
173 }
174 }
175 }
176
177 forall_goto_program_instructions(it, goto_function.body)
178 {
179 auto i = cp_map.find(&*it);
180 if(i == cp_map.end())
181 continue;
182 auto inst = i->first;
183 if(is_cutpoint(*inst))
184 continue;
185
186 m_cps.push_back(std::make_shared<cutpointt>(*this, m_cps.size(), *inst));
187 m_insts.insert(std::make_pair(inst, m_cps.back()));
188 }
189}

191void cutpoint_grapht::compute_fwd_reach(const goto_functiont &goto_function)
192{
193 for(auto it = goto_function.body.instructions.rbegin();
194 it != goto_function.body.instructions.rend();)
195 {
196 auto itf = (++it).base();
197 auto succs = goto_function.body.get_successors(itf);
198 // if (succs.empty()) continue;
199 auto inst = &*itf;
200 m_fwd.insert(std::make_pair(inst, reachabilityt()));
201 reachabilityt &r = m_fwd[inst];
202 for(auto succ : succs)
203 {
204 if(is_cutpoint(*succ))
205 r.set(get_cutpoint(*succ).id());
206 else
207 r |= m_fwd[&*succ];
208 }
209 }
